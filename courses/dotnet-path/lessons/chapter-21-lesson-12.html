در این بخش همه مفاهیم فصل را در یک Payment Domain Model کنار هم می‌گذاریم.

Value Object: Money

C#
public sealed record Money
{
    public decimal Amount { get; }
    public string Currency { get; }

    public Money(
        decimal amount,
        string currency)
    {
        if (amount <= 0)
            throw new ArgumentOutOfRangeException(
                nameof(amount));

        if (string.IsNullOrWhiteSpace(currency))
            throw new ArgumentException(
                "Currency is required.");

        Amount = amount;
        Currency = currency;
    }
}

Entity داخلی: Refund

C#
public class Refund
{
    public Guid Id { get; }
    public Money Amount { get; }

    private Refund(
        Guid id,
        Money amount)
    {
        Id = id;
        Amount = amount;

    public static Refund Create(
        Guid id,
        Money amount)

Aggregate Root: Payment

C#
public class Payment
{
    private readonly List<Refund>
        _refunds = new();

    private readonly List<object>
        _domainEvents = new();

    public Guid Id { get; }
    public Money Amount { get; }

    public PaymentStatus Status
    {
        get;
        private set;
    }

    public IReadOnlyCollection<Refund>
        Refunds => _refunds.AsReadOnly();

    public IReadOnlyCollection<object>
        DomainEvents => _domainEvents.AsReadOnly();

    public Payment(
        Guid id,
        Money amount)
    {
        Id = id;
        Amount = amount;
        Status = PaymentStatus.Pending;
    }

    public void Authorize()
    {
        if (Status != PaymentStatus.Pending)
            throw new InvalidOperationException();

        Status = PaymentStatus.Authorized;
    }

    public void Capture()
    {
        if (Status != PaymentStatus.Authorized)
            throw new InvalidOperationException();

        Status = PaymentStatus.Paid;

        _domainEvents.Add(
            new PaymentCaptured(
                Id,
                Amount,
                DateTime.UtcNow));
    }

    public void RequestRefund(
        Money refundAmount)
    {
        if (Status != PaymentStatus.Paid)
            throw new InvalidOperationException(
                "Only paid payment can be refunded.");

        if (refundAmount.Currency
            != Amount.Currency)
            throw new InvalidOperationException(
                "Currency mismatch.");

        var totalRefunded =
            _refunds.Sum(
                x => x.Amount.Amount);

        if (totalRefunded
            + refundAmount.Amount
            > Amount.Amount)
            throw new InvalidOperationException(
                "Refund exceeds payment amount.");

        _refunds.Add(
            Refund.Create(
                Guid.NewGuid(),
                refundAmount));
    }
}

Domain Service

C#
public class PaymentFeeService
{
    public Money Calculate(
        Merchant merchant,
        Money amount)
    {
        var rate =
            merchant.IsVip
                ? 0.005m
                : 0.01m;

        return new Money(
            amount.Amount * rate,
            amount.Currency);
    }
}

Repository Contract

C#
public interface IPaymentRepository
{
    Task<Payment?> GetAsync(
        Guid paymentId,
        CancellationToken cancellationToken);

    Task SaveAsync(
        Payment payment,
        CancellationToken cancellationToken);
}

Application Use Case

C#
public class CapturePaymentHandler
{
    private readonly IPaymentRepository
        _repository;

    public CapturePaymentHandler(
        IPaymentRepository repository)
    {
        _repository = repository;
    }

    public async Task HandleAsync(
        Guid paymentId,
        CancellationToken cancellationToken)
    {
        var payment =
            await _repository.GetAsync(
                paymentId,
                cancellationToken)
            ?? throw new InvalidOperationException(
                "Payment not found.");

        payment.Capture();

        await _repository.SaveAsync(
            payment,
            cancellationToken);
    }
}

جمع‌بندی فصل 21

در DDD، OOP برای مدل‌کردن Business استفاده می‌شود: Entity هویت دارد، Value Object با مقدار تعریف می‌شود، Aggregate مرز Consistency است، Aggregate Root از Invariantها محافظت می‌کند و Domain Event اتفاق مهم Business را بیان می‌کند.
