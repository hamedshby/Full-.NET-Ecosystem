Domain Entity

C#
public class Payment
{
    public Guid Id { get; private set; }

    public decimal Amount { get; private set; }

    public decimal Fee { get; private set; }

    public PaymentStatus Status { get; private set; }

    private Payment()
    {
    }

    public static Payment Create(
        decimal amount,
        decimal fee)
    {
        return new Payment
        {
            Id = Guid.NewGuid(),
            Amount = amount,
            Fee = fee,
            Status = PaymentStatus.Pending
        };
    }

    public void MarkAsSuccessful()
    {
        Status =
            PaymentStatus.Successful;
    }

    public void MarkAsFailed()
    {
        Status =
            PaymentStatus.Failed;
    }
}

Application Use Case نهایی

C#
public class ProcessPaymentHandler
{
    private readonly IPaymentValidator
        _validator;

    private readonly IFeeCalculator
        _feeCalculator;

    private readonly IPaymentGateway
        _gateway;

    private readonly IPaymentRepository
        _repository;

    private readonly IEventPublisher
        _publisher;

    private readonly IPaymentNotifier
        _notifier;

    public ProcessPaymentHandler(
        IPaymentValidator validator,
        IFeeCalculator feeCalculator,
        IPaymentGateway gateway,
        IPaymentRepository repository,
        IEventPublisher publisher,
        IPaymentNotifier notifier)
    {
        _validator = validator;
        _feeCalculator = feeCalculator;
        _gateway = gateway;
        _repository = repository;
        _publisher = publisher;
        _notifier = notifier;
    }

    public async Task<PaymentResult> HandleAsync(
        PaymentRequest request,
        CancellationToken cancellationToken)
    {
        _validator.Validate(
            request);

        decimal fee =
            _feeCalculator.Calculate(
                request.Amount);

        var payment =
            Payment.Create(
                request.Amount,
                fee);

        var gatewayResult =
            await _gateway.PayAsync(
                new PaymentGatewayRequest(
                    request.Amount + fee),
                cancellationToken);

        if (!gatewayResult.IsSuccess)
        {
            payment.MarkAsFailed();

            await _repository.AddAsync(
                payment,
                cancellationToken);

            return PaymentResult.Failed(
                gatewayResult.ErrorCode);
        }

        payment.MarkAsSuccessful();

        await _repository.AddAsync(
            payment,
            cancellationToken);

        await _publisher.PublishAsync(
            new PaymentCompleted(
                payment.Id,
                payment.Amount),
            cancellationToken);

        await _notifier.NotifyAsync(
            payment,
            cancellationToken);

        return PaymentResult.Success(
            payment.Id,
            gatewayResult.ReferenceNumber);
    }
}

آیا Handler خودش SRP را نقض نمی‌کند؟

خیر، چون Responsibility اصلی آن Orchestration یک Use Case مشخص است. Ruleهای تخصصی در Componentهای جدا قرار دارند.

public record PaymentResult
{
    public bool IsSuccess { get; init; }

    public Guid? PaymentId { get; init; }

    public string? ReferenceNumber { get; init; }

    public string? ErrorCode { get; init; }

    public static PaymentResult Success(
        Guid paymentId,
        string? referenceNumber)
        => new()
        {
            IsSuccess = true,
            PaymentId = paymentId,
            ReferenceNumber = referenceNumber
        };

    public static PaymentResult Failed(
        string? errorCode)
        => new()
        {
            IsSuccess = false,
            ErrorCode = errorCode
        };
}

هر اصل SOLID کجاست؟

اصل در پروژه
SRP Validator، FeeCalculator، Repository، Gateway، Publisher و Notifier مسئولیت‌های جدا دارند.
OCP Gateway جدید با Implementation جدید اضافه می‌شود.
LSP تمام Gatewayها Contract یکسان IPaymentGateway را حفظ می‌کنند.
ISP Purchase و Refund در Contractهای جدا قرار می‌گیرند.
DIP Application به Abstraction وابسته است و Infrastructure آن را implement می‌کند.

آیا این طراحی Over-Engineering نیست؟

در یک پروژه خیلی کوچک شاید باشد. SOLID به معنی ساخت Interface و Class برای هر چند خط کد نیست. در نقاطی از سیستم که واقعاً Change و Variation وجود دارد، این تفکیک ارزشمند می‌شود.

⚠️
هدف SOLID افزایش تعداد فایل‌ها نیست؛ هدف کاهش Coupling، محدود کردن دامنه تغییر و واضح‌تر کردن Responsibilityهاست.

Code Review Checklist

سؤال اصل
آیا Class چند Responsibility مستقل دارد؟ SRP
برای Feature جدید باید Class پایدار را تغییر دهیم؟ OCP
آیا تمام Subtypeها Contract یکسان دارند؟ LSP
آیا Client به Method غیرضروری وابسته است؟ ISP
آیا Business Logic مستقیماً Infrastructure را می‌شناسد؟ DIP

پاسخ مصاحبه‌ای کوتاه

English
In a payment service, I separate validation, persistence,
gateway communication, notification, and event publishing.

The application layer depends on abstractions such as
IPaymentGateway and IPaymentRepository.

New gateways can be added without changing the payment use case,
which supports OCP.

All gateway implementations follow the same contract,
which supports LSP.

Payment and refund capabilities can be separated into focused interfaces,
which supports ISP.

And the business logic remains independent from SQL Server,
RabbitMQ, and specific bank providers,
which is an example of DIP.

جمع‌بندی مثال کامل SOLID

SOLID زمانی ارزش واقعی خود را نشان می‌دهد که سیستم رشد می‌کند و تغییرات متعدد وارد می‌شوند. یک طراحی خوب باید اجازه دهد Database، Message Broker، Payment Gateway یا Business Rule تغییر کند، بدون اینکه بخش‌های نامرتبط سیستم بازنویسی شوند.

✅
نکته کلیدی: SOLID درباره تعداد Classها نیست؛ درباره ساخت مرزهایی است که تغییر را قابل کنترل می‌کنند.
